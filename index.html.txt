    <script>
        // ⚠️ ใส่ BIN ID และ MASTER KEY จาก jsonbin.io ตรงนี้ครับ
        const BIN_ID = "ใส่_BIN_ID_ตรงนี้";
        const MASTER_KEY = "ใส่_MASTER_KEY_ตรงนี้";

        let scriptsData = [];
        let ownerPassword = localStorage.getItem('redOwnerPass') || "1234";

        function changeOwnerPassword() {
            const currentPass = prompt("กรุณากรอกรหัสผ่านปัจจุบัน (ค่าเริ่มต้นคือ 1234):");
            if (currentPass === ownerPassword) {
                const newPass = prompt("ตั้งรหัสผ่านเจ้าของใหม่:");
                if (newPass && newPass.trim() !== "") {
                    ownerPassword = newPass.trim();
                    localStorage.setItem('redOwnerPass', ownerPassword);
                    alert("เปลี่ยนรหัสผ่านเจ้าของเรียบร้อยแล้ว!");
                }
            } else if (currentPass !== null) {
                alert("รหัสผ่านไม่ถูกต้อง!");
            }
        }

        function loadScripts() {
            fetch(`https://api.jsonbin.io/v3/b/${BIN_ID}/latest`, {
                headers: { 'X-Master-Key': MASTER_KEY }
            })
            .then(res => res.json())
            .then(data => {
                scriptsData = data.record || [];
                renderScripts();
            })
            .catch(err => {
                console.error(err);
                document.getElementById('scriptList').innerHTML = '<div class="card empty-box">โหลดข้อมูลไม่สำเร็จ ตรวจสอบ BIN ID / Key</div>';
            });
        }

        function saveScriptsOnline(alertText) {
            const btn = document.getElementById('submitBtn');
            btn.innerText = "กำลังบันทึก...";
            btn.disabled = true;

            fetch(`https://api.jsonbin.io/v3/b/${BIN_ID}`, {
                method: 'PUT',
                headers: {
                    'Content-Type': 'application/json',
                    'X-Master-Key': MASTER_KEY
                },
                body: JSON.stringify(scriptsData)
            })
            .then(res => res.json())
            .then(() => {
                btn.innerText = "ลงผลงาน";
                btn.disabled = false;
                alert(alertText);
                renderScripts();
            })
            .catch(err => {
                alert("เกิดข้อผิดพลาดในการบันทึก");
                btn.innerText = "ลงผลงาน";
                btn.disabled = false;
            });
        }

        function renderScripts() {
            const list = document.getElementById('scriptList');
            list.innerHTML = '';

            if (scriptsData.length === 0) {
                list.innerHTML = '<div class="card empty-box">ยังไม่มีผลงานสคริปต์ในระบบ</div>';
                return;
            }

            scriptsData.forEach((item, index) => {
                const div = document.createElement('div');
                div.className = 'script-item';
                
                const imgHTML = item.imgData ? `<img src="${item.imgData}" class="script-img">` : '';

                div.innerHTML = `
                    <div class="script-header">
                        <div class="script-title">${cleanHTML(item.title)}</div>
                        <div class="action-btns">
                            <button class="btn-act btn-copy" onclick="copyCode(${index})">📋 คัดลอก</button>
                            <button class="btn-act btn-delete" onclick="deleteScript(${index})">🗑️ ลบ</button>
                        </div>
                    </div>
                    ${imgHTML}
                    <div class="script-code">${cleanHTML(item.code)}</div>
                `;
                list.appendChild(div);
            });
        }

        function cleanHTML(str) {
            return str ? str.replace(/[&<>'"]/g, tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)) : '';
        }

        // แปลงไฟล์รูปภาพในเครื่องเป็น Base64 Data String
        function processAndSave(title, code, file) {
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    const imgData = e.target.result;
                    scriptsData.unshift({ title, imgData, code });
                    saveScriptsOnline('ลงผลงานพร้อมรูปภาพสำเร็จ!');
                };
                reader.readAsDataURL(file);
            } else {
                scriptsData.unshift({ title, imgData: null, code });
                saveScriptsOnline('ลงผลงานสำเร็จ!');
            }
        }

        document.getElementById('scriptForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const title = document.getElementById('title').value;
            const code = document.getElementById('code').value;
            const fileInput = document.getElementById('imgFile');
            const file = fileInput.files[0];

            processAndSave(title, code, file);
            
            document.getElementById('title').value = '';
            document.getElementById('code').value = '';
            fileInput.value = '';
        });

        function copyCode(index) {
            navigator.clipboard.writeText(scriptsData[index].code).then(() => {
                alert('คัดลอกสคริปต์เรียบร้อยแล้ว!');
            });
        }

        function deleteScript(index) {
            const inputPass = prompt("🔒 ใส่รหัสผ่านเจ้าของเพื่อลบสคริปต์:");
            if (inputPass === ownerPassword) {
                scriptsData.splice(index, 1);
                saveScriptsOnline('ลบสคริปต์เรียบร้อยแล้ว!');
            } else if (inputPass !== null) {
                alert("❌ รหัสผ่านไม่ถูกต้อง!");
            }
        }

        loadScripts();
    </script>
